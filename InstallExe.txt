# 1. Clone
git clone <repo-url>
cd cv-pii-tool

# 2. Python 3.11+ is required — check first
python --version

# 3. Create virtual environment
python -m venv .venv

# Windows:
.venv\Scripts\activate
# Mac/Linux:
source .venv/bin/activate

# 4. Install dependencies
pip install -r requirements.txt

# 5. Verify
pytest tests/test_sanitiser.py -q

# 6. Drop CVs in and run
# (copy your CV files into input/)
python -m src.runner --input input --output output --ner heuristic
